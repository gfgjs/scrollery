cmake_minimum_required(VERSION 3.20)
project(scrollery_vpl_bridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "VPL bridge requires Windows x64")
endif()
find_package(VPL 2.17.0 EXACT REQUIRED COMPONENTS dispatcher)
add_library(scrollery_vpl_bridge STATIC bridge.cpp)
target_compile_features(scrollery_vpl_bridge PRIVATE cxx_std_17)
target_compile_definitions(scrollery_vpl_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(scrollery_vpl_bridge PRIVATE /W4 /WX /utf-8)
target_link_libraries(scrollery_vpl_bridge PUBLIC VPL::dispatcher d3d11 dxgi)
install(TARGETS scrollery_vpl_bridge ARCHIVE DESTINATION lib)
option(SCROLLERY_VPL_SMOKE "Build the targeted bridge smoke check" OFF)
if(SCROLLERY_VPL_SMOKE)
  add_executable(scrollery_vpl_smoke smoke.cpp)
  target_compile_features(scrollery_vpl_smoke PRIVATE cxx_std_17)
  target_compile_definitions(scrollery_vpl_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(scrollery_vpl_smoke PRIVATE /W4 /WX /utf-8)
  target_link_libraries(scrollery_vpl_smoke PRIVATE scrollery_vpl_bridge windowscodecs ole32)
endif()
